function saveSale(){ const customer = document.getElementById('sCustomer').value.trim(); const desc = document.getElementById('sDesc').value.trim(); if(!customer && !desc){ return alert('Please enter Customer Name or Description / Item.'); } const amountValue = document.getElementById('sAmount').value.trim(); let amount = amountValue === '' ? 0 : Number(amountValue); const type = document.getElementById('sType').value; const advance = type === 'credit' ? Number(document.getElementById('sAdvance').value || 0) : 0; const enteredBalance = Number( document.getElementById('sBalancePayment').value || 0 ); const balancePayment = type === 'credit' ? enteredBalance : 0; if(type === 'balance_payment' && enteredBalance > 0){ amount = enteredBalance; } if(!Number.isFinite(amount) || amount < 0){ return alert('Please enter a valid amount.'); } if(!Number.isFinite(advance) || advance < 0){ return alert('Please enter a valid Advance Payment.'); } if(!Number.isFinite(enteredBalance) || enteredBalance < 0){ return alert('Please enter a valid Balance Payment.'); } if(type === 'credit' && advance + balancePayment > amount){ return alert('Advance Payment + Balance Payment cannot be greater than Credit Amount.'); } if(type === 'balance_payment' && amount <= 0){ return alert('Please enter the balance payment amount.'); } const obj = { type, amount, customer, desc, date: document.getElementById('sDate').value, time: document.getElementById('sTime').value || nowLocal().time, advance_payment: advance, balance_payment: balancePayment }; if(editingSale){ const x = data.sales.find(s => s.id === editingSale); if(x) Object.assign(x, obj); } else { data.sales.push({id: uid(), ...obj}); } persist(); closeSaleModal(); renderDashboard(); renderSales(); }
